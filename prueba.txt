LONGINT variable_entera, contador;

FLOAT resultado;

// 2. Declaración de Typedef (Tema 23)

// 4. Declaración de Función
FLOAT FUNCTION calcular_doble(FLOAT parametro)
    FLOAT temporal;
BEGIN
    temporal := parametro * 2.0;
    RET(temporal);
END;

// ==========================================
// BLOQUE DE EJECUCIÓN (BEGIN - END principal)
// ==========================================
BEGIN
    
    // 5. Asignaciones y constantes propias de tu Lexer
    variable_entera := 15$l;
    variable_flotante := 12.5s2;
    resultado := 0.0;

    // 6. Sentencia de Salida
    POUT("Iniciando la prueba del analizador sintactico exitosa");

    // 7. Expresiones matemáticas complejas
    resultado := variable_entera + 5$l * 2$l;

    // 8. Estructura IF - ELSE
    IF (resultado >= 10.0)
    BEGIN
        POUT("El resultado es mayor o igual a 10");
        resultado := resultado - 1.0;
    END
    ELSE
    BEGIN
        POUT("El resultado es menor");
    END
    END_IF;

    // 9. Estructura REPEAT UNTIL
    contador := 0$l;
    REPEAT
    BEGIN
        contador := contador + 1$l;
    END
    UNTIL (contador == 5$l);

    // 10. Casting TOSF
    resultado := TOSF(variable_entera);

    // 11. Invocación de función (con orden de evaluación del Tema 20)
    resultado := calcular_doble(resultado) [1, 2];

END
