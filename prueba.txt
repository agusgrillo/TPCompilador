// ==========================================

// ==========================================

// 1. Declaración de variables simples
LONGINT variable_entera, contador;
SINGLEF variable_flotante, resultado;

// 2. Declaración de Typedef (Tema 23)
// Aca usamos el sufijo $l porque la lista de valores exige números enteros validos
TYPEDEF mis_estados = [1$l, 2$l, 3$l];

// 3. Declaración de Clase 
// IMPORTANTE: Escrita 100% en minúsculas porque así lo exige la regla de Identificadores
CLASS mi_clase BEGIN
    LONGINT atributo_clase;
END;

// 4. Declaración de Función
SINGLEF FUNCTION calcular_doble(SINGLEF parametro)
    SINGLEF temporal;
BEGIN
    temporal := parametro * 2.0;
    RET(temporal);
END;

// ==========================================
// BLOQUE DE EJECUCIÓN (BEGIN - END principal)
// ==========================================
BEGIN
    
    // 5. Asignaciones y constantes 
    variable_entera := 15$l;
    variable_flotante := 12.5s2;
    resultado := 0.0;

    // 6. Sentencia de Salida
    POUT("Iniciando la prueba del analizador sintactico exitosa");

    // 7. Expresiones matemáticas complejas
    resultado := variable_flotante + 5.0 * 2.0;

    // 8. Asignación de expresiones en expresiones (Tema 17)
    variable_entera := contador = (1$l + 2$l);

    // 9. Estructura IF - ELSE
    IF (resultado >= 10.0)
    BEGIN
        POUT("El resultado es mayor o igual a 10");
        resultado := resultado - 1.0;
    END
    ELSE
    BEGIN
        POUT("El resultado es menor");
    END
    END_IF;

    // 10. Estructura REPEAT UNTIL
    contador := 0$l;
    REPEAT
    BEGIN
        contador := contador + 1$l;
    END
    UNTIL (contador == 5$l);

    // 11. Casting TOSF
    resultado := TOSF(variable_entera);

    // 12. Invocación de función con orden de evaluación del Tema 20
    // Llevan $l porque el corchete recibe lista_valores, que son constantes NUMBER
    resultado := calcular_doble(resultado) [1$l, 2$l];

END
