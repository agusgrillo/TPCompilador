    // Error 9: Ausencia de lista de clases despues de EXTENDS (Temas 30-32)

// BLOQUE PRINCIPAL EJECUTABLE

BEGIN
    // Error 10: Falta ';' al final de una sentencia ejecutable
    val_a := 10.0;

    // Error 11: Falta de operando en expresion matematica

    // Error 12: Falta de operador en expresion matematica
    val_b := 5.0 2.0;

    // Error 13: Uso de ':=' donde debe usarse '=' en expresion (Temas 17 y 18)
    val_a := val_b := (3.0);

    // Error 14: Falta de argumento en sentencia POUT
    POUT();

    // Error 15: Falta ';' al final de la sentencia IF
    IF (val_a > 0.0)
    BEGIN
        val_a := val_a - 1.0;
    END
    END_IF

    // Error 16: Falta la palabra reservada 'UNTIL' en REPEAT (Tema 12)
    REPEAT
    BEGIN
        val_a := val_a + 1.0;
    END
    (val_a == 5.0);

    POUT("Fin del test de errores");
